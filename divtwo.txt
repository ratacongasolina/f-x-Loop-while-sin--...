    if (x < 0):                                    #n= Schleifenzähler
        y = 0
    else:
        for n in range (0, x):                     #for = geht jede Einheit der Zahl x nacheinander durch.
            if (d == 0):                           #wenn d auf 0 steht, wird das erste Element eines neuen Paares gefunden.
                d = 1                              #d =1 = halbes paar
